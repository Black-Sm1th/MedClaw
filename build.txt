==============================================================
MedClaw 在 ARM64 (aarch64) Linux 上打包 AppImage 流程
==============================================================

前提：
- 目标架构：aarch64 (arm64)
- 推荐发行版：Ubuntu 20.04 / 22.04 (arm64) 或 Debian 11/12 (arm64)
- 操作建议在与目标机器相同发行版的真机 / 虚拟机 / 树莓派 / 飞腾·鲲鹏机器
  上完成。glibc 版本必须 ≤ 目标机，否则在更老系统上会报
  `GLIBC_X.YY not found`。
- 全程不需要 root 也能完成，但安装系统包要 sudo。

==============================================================
0. 系统包准备 (一次性)
==============================================================

sudo apt-get update
sudo apt-get install -y \
    build-essential git curl wget pkg-config \
    libgl1-mesa-dev libglu1-mesa-dev \
    libxkbcommon-dev libxkbcommon-x11-dev \
    libfontconfig1-dev libfreetype6-dev \
    libx11-dev libx11-xcb-dev libxext-dev libxfixes-dev \
    libxi-dev libxrender-dev libxcb1-dev libxcb-glx0-dev \
    libxcb-keysyms1-dev libxcb-image0-dev libxcb-shm0-dev \
    libxcb-icccm4-dev libxcb-sync-dev libxcb-xfixes0-dev \
    libxcb-shape0-dev libxcb-randr0-dev libxcb-render-util0-dev \
    libxcb-util-dev libxcb-xinerama0-dev libxcb-xkb-dev \
    libxcb-cursor-dev libdbus-1-dev libssl-dev \
    patchelf file fuse libfuse2

# fuse / libfuse2 用于直接运行 AppImage（包含 appimagetool 自身）
# 如果系统没有 libfuse2（如新版 Ubuntu），运行 AppImage 时改用：
#     ./xxx.AppImage --appimage-extract-and-run

==============================================================
1. 安装 Qt 5.15.x (aarch64) —— 三选一
==============================================================

【方式 A：系统 qt5（最省事，版本可能略有差异）】
  sudo apt-get install -y \
      qtbase5-dev qtbase5-dev-tools \
      qtdeclarative5-dev qtquickcontrols2-5-dev \
      qtwebsockets5-dev qml-module-qtquick2 \
      qml-module-qtquick-controls2 qml-module-qtquick-window2 \
      qml-module-qtquick-layouts qml-module-qtquick-dialogs \
      qml-module-qt-labs-platform qml-module-qt-labs-settings \
      qml-module-qtgraphicaleffects qml-module-qtquick-templates2 \
      qtchooser qtcreator
  # 之后 qmake 路径就是 /usr/bin/qmake (qt5)
  # 如需固定为 qt5：export QT_SELECT=qt5

【方式 B：用 aqtinstall 拉官方 Qt 5.15.2 桌面包（需有 ARM64 版本）】
  # 官方 5.15.2 桌面 Linux 没有 arm64 预编译包，但 6.5+ 有。
  # 如果改用 Qt 6 自行打包，参考 Qt6 文档，注意 .pro 兼容性。
  python3 -m pip install -U aqtinstall
  python3 -m aqt install-qt linux desktop 6.5.3 linux_gcc_arm64 \
      -m qtwebsockets qtdeclarative qtshadertools \
      -O ~/Qt
  # 之后 qmake：~/Qt/6.5.3/gcc_arm64/bin/qmake

【方式 C：从源码编译 Qt 5.15.x（耗时长，最干净）】
  git clone https://code.qt.io/qt/qt5.git ~/qt5-src
  cd ~/qt5-src
  git checkout 5.15.2
  perl init-repository --module-subset=qtbase,qtdeclarative,qtquickcontrols2,qtwebsockets,qtgraphicaleffects,qtimageformats,qtsvg
  mkdir -p ~/qt5-build && cd ~/qt5-build
  ~/qt5-src/configure -opensource -confirm-license -release \
      -prefix $HOME/Qt/5.15.2-aarch64 -nomake examples -nomake tests
  make -j$(nproc) && make install
  # 之后 qmake：~/Qt/5.15.2-aarch64/bin/qmake

下文以变量 QT_DIR 指代 Qt 安装目录，自行替换：
  方式 A:  QT_DIR=/usr   (qmake 在 /usr/bin/qmake)
  方式 B/C: QT_DIR=~/Qt/<version>/<arch_dir>

  export PATH="$QT_DIR/bin:$PATH"
  qmake --version    # 确认输出 aarch64 版本的 Qt

==============================================================
2. 编译 patchelf / linuxdeployqt （ARM64 没有官方二进制）
==============================================================

# patchelf：方式 1 直接 apt
sudo apt-get install -y patchelf
patchelf --version    # 0.14+ 即可

# linuxdeployqt 官方只发布 x86_64 二进制，aarch64 必须自行编译
git clone https://github.com/probonopd/linuxdeployqt.git ~/linuxdeployqt-src
cd ~/linuxdeployqt-src
mkdir build && cd build
$QT_DIR/bin/qmake ../linuxdeployqt.pro
make -j$(nproc)
sudo cp ./bin/linuxdeployqt /usr/local/bin/linuxdeployqt
linuxdeployqt --version

==============================================================
3. 准备 appimagetool (aarch64)
==============================================================

mkdir -p ~/.local/bin
curl -L --fail -o ~/.local/bin/appimagetool \
    https://github.com/AppImage/appimagetool/releases/download/continuous/appimagetool-aarch64.AppImage
chmod +x ~/.local/bin/appimagetool
export PATH="$HOME/.local/bin:$PATH"
appimagetool --version

==============================================================
4. 编译 MedClaw (Release)
==============================================================

cd /path/to/MedClaw
rm -rf build/release-arm64
mkdir -p build/release-arm64
cd build/release-arm64

$QT_DIR/bin/qmake CONFIG+=release ../../MedClaw.pro
make -j$(nproc)

# 产出：build/release-arm64/MedClaw  (aarch64 ELF)
file MedClaw   # 应包含 'ARM aarch64'

==============================================================
5. 构建 AppDir 目录结构
==============================================================

cd /path/to/MedClaw
rm -rf build/AppDir
mkdir -p build/AppDir/usr/bin
mkdir -p build/AppDir/usr/share/applications
mkdir -p build/AppDir/usr/share/icons/hicolor/256x256/apps
mkdir -p build/AppDir/usr/share/doc/libc6

cp build/release-arm64/MedClaw build/AppDir/usr/bin/MedClaw
strip --strip-unneeded build/AppDir/usr/bin/MedClaw

cp images/256*256.png build/AppDir/usr/share/icons/hicolor/256x256/apps/medclaw.png

# linuxdeployqt 在 -unsupported-allow-new-glibc 模式下会强制
# 检查这个文件存在；本机的 libc6 copyright 可直接复用
cp /usr/share/doc/libc6/copyright build/AppDir/usr/share/doc/libc6/copyright \
    || echo "libc6 copyright placeholder" > build/AppDir/usr/share/doc/libc6/copyright

# desktop 文件
cat > build/AppDir/usr/share/applications/medclaw.desktop <<'EOF'
[Desktop Entry]
Type=Application
Name=MedClaw
Comment=MedClaw client
Exec=MedClaw
Icon=medclaw
Categories=Network;Office;
Terminal=false
StartupWMClass=MedClaw
EOF

==============================================================
6. 收集 Qt / QML 依赖
==============================================================

export PATH="$QT_DIR/bin:$PATH"

linuxdeployqt build/AppDir/usr/share/applications/medclaw.desktop \
    -qmake=$QT_DIR/bin/qmake \
    -qmldir=. \
    -extra-plugins=iconengines,imageformats,platforminputcontexts \
    -bundle-non-qt-libs \
    -unsupported-allow-new-glibc \
    -verbose=1

# 校验：build/AppDir/usr/lib 应含 libQt5Core.so.5 等
#       build/AppDir/usr/qml 应含 QtQuick / QtGraphicalEffects
#       build/AppDir/usr/plugins 应含 platforms/libqxcb.so

==============================================================
7. 生成 AppImage（aarch64）
==============================================================

cd /path/to/MedClaw
ARCH=aarch64 appimagetool build/AppDir build/MedClaw-aarch64.AppImage

# 不带 FUSE 的环境改用：
#   ARCH=aarch64 appimagetool --appimage-extract-and-run \
#       build/AppDir build/MedClaw-aarch64.AppImage

ls -lh build/MedClaw-aarch64.AppImage
file  build/MedClaw-aarch64.AppImage   # 应该是 ELF 64-bit aarch64

==============================================================
8. 验证
==============================================================

# 直接运行
./build/MedClaw-aarch64.AppImage

# 没装 FUSE 的目标机
./build/MedClaw-aarch64.AppImage --appimage-extract-and-run

# 排错时输出更多日志
QT_DEBUG_PLUGINS=1 ./build/MedClaw-aarch64.AppImage

==============================================================
常见问题
==============================================================

(1) linuxdeployqt 在 "Found icons from desktop file" 后静默 exit 1
    原因：开启 -unsupported-allow-new-glibc 时，源码强制要求
        AppDir/usr/share/doc/libc6/copyright 存在，否则 exit(1)。
    解决：第 5 步已补占位文件。

(2) ERROR: Could not start patchelf
    解决：apt-get install -y patchelf

(3) qt.network.ssl: cannot resolve EVP_PKEY_base_id 等
    原因：Qt 5.15.2 期望 OpenSSL 1.1.x，但运行环境是 OpenSSL 3.x。
    若仅用 ws:// 可忽略；需要 wss:// 时把 libssl.so.1.1 / libcrypto.so.1.1
    复制到 AppDir/usr/lib/ 一同分发。

(4) "could not load the Qt platform plugin xcb"
    多半缺少 xcb 系列运行时；apt-get install libxcb-cursor0 libxkbcommon-x11-0
    再确认 platforms/libqxcb.so 已被打进 AppDir/usr/plugins/platforms/。

(5) GLIBC_X.YY not found
    打包机 glibc 比目标机新。请在与目标系统同代的 ARM64 机器
    （或 Docker：arm64v8/ubuntu:20.04）上重新执行 1~7 步。

(6) 想做交叉编译（在 x86_64 上构建 aarch64）
    建议改用 docker：
        docker run --rm --platform=linux/arm64 \
            -v $PWD:/work -w /work arm64v8/ubuntu:22.04 bash
    在容器内执行 0~7 步，QEMU 透明仿真即可。性能远低于真机，
    但脚本无需任何修改。

==============================================================
一键脚本（参考，按需保存为 build_arm64.sh）
==============================================================

#!/usr/bin/env bash
set -euo pipefail
QT_DIR="${QT_DIR:-/usr}"          # 如用方式 B/C 自行覆盖
PROJECT_ROOT="$(cd "$(dirname "$0")" && pwd)"
cd "$PROJECT_ROOT"

export PATH="$QT_DIR/bin:$HOME/.local/bin:$PATH"

rm -rf build/release-arm64 build/AppDir
mkdir -p build/release-arm64
( cd build/release-arm64 && qmake CONFIG+=release ../../MedClaw.pro && make -j"$(nproc)" )

mkdir -p build/AppDir/usr/{bin,share/applications,share/icons/hicolor/256x256/apps,share/doc/libc6}
cp build/release-arm64/MedClaw build/AppDir/usr/bin/MedClaw
strip --strip-unneeded build/AppDir/usr/bin/MedClaw
cp images/256*256.png build/AppDir/usr/share/icons/hicolor/256x256/apps/medclaw.png
cp /usr/share/doc/libc6/copyright build/AppDir/usr/share/doc/libc6/copyright \
    || echo placeholder > build/AppDir/usr/share/doc/libc6/copyright

cat > build/AppDir/usr/share/applications/medclaw.desktop <<'EOF'
[Desktop Entry]
Type=Application
Name=MedClaw
Comment=MedClaw client
Exec=MedClaw
Icon=medclaw
Categories=Network;Office;
Terminal=false
StartupWMClass=MedClaw
EOF

linuxdeployqt build/AppDir/usr/share/applications/medclaw.desktop \
    -qmake="$QT_DIR/bin/qmake" \
    -qmldir=. \
    -extra-plugins=iconengines,imageformats,platforminputcontexts \
    -bundle-non-qt-libs \
    -unsupported-allow-new-glibc \
    -verbose=1

ARCH=aarch64 appimagetool build/AppDir build/MedClaw-aarch64.AppImage
echo "Done -> build/MedClaw-aarch64.AppImage"
